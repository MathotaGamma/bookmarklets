(function () {
  try {
    var owner = 'MathotaGamma';
    var repo = 'extensions';
    var dir = 'scripts';
    var branch = '';
    var file = 'script.js';

    dir = dir.replace(/^\/+|\/+$/g, '');
    file = file.replace(/^\/+/, '');
    var path = (dir ? dir + '/' : '') + file;
    var br = branch || 'HEAD';
    var PROBE = 'data-file-runner-probe';

    // パスを区切りごとにURLエンコード(スペースや日本語を含む名前に対応)
    function enc(p) {
      return p.split('/').map(encodeURIComponent).join('/');
    }

    // <script> にコードを入れて実行できるか、事前に確かめる
    function canRunInline() {
      var root = document.documentElement;
      root.removeAttribute(PROBE);
      var s = document.createElement('script');
      s.textContent = "document.documentElement.setAttribute('" + PROBE + "','1')";
      root.appendChild(s);
      s.remove();
      var ok = root.getAttribute(PROBE) === '1';
      root.removeAttribute(PROBE);
      return ok;
    }

    function runInline(code) {
      var s = document.createElement('script');
      s.textContent = code;
      document.documentElement.appendChild(s);
      s.remove();
    }

    // Blob の <script> として実行。読み込めたら true
    function runBlob(code) {
      return new Promise(function (resolve) {
        var url = URL.createObjectURL(new Blob([code], { type: 'text/javascript' }));
        var s = document.createElement('script');
        function done(ok) {
          s.remove();
          URL.revokeObjectURL(url);
          resolve(ok);
        }
        s.onload = function () {
          done(true);
        };
        s.onerror = function () {
          done(false);
        };
        s.src = url;
        document.documentElement.appendChild(s);
      });
    }

    // jsDelivr から <script> で読み込む。読み込めたら true
    function runFromCdn() {
      var getBranch = branch
        ? Promise.resolve(branch)
        : fetch('https://api.github.com/repos/' + owner + '/' + repo)
            .then(function (r) {
              return r.ok ? r.json() : {};
            })
            .then(function (d) {
              return d.default_branch || 'main';
            })
            .catch(function () {
              return 'main';
            });
      return getBranch.then(function (b) {
        return new Promise(function (resolve) {
          var s = document.createElement('script');
          s.onload = function () {
            resolve(true);
          };
          s.onerror = function () {
            s.remove();
            resolve(false);
          };
          s.src = 'https://cdn.jsdelivr.net/gh/' + owner + '/' + repo + '@' + b + '/' + enc(path);
          document.documentElement.appendChild(s);
        });
      });
    }

    fetch('https://raw.githubusercontent.com/' + owner + '/' + repo + '/' + br + '/' + enc(path))
      .then(function (r) {
        if (!r.ok) throw new Error('HTTP ' + r.status);
        return r.text();
      })
      .then(function (code) {
        if (canRunInline()) {
          runInline(code);
          return true;
        }
        return runBlob(code);
      })
      .catch(function () {
        return false;
      })
      .then(function (ok) {
        return ok || runFromCdn();
      })
      .then(function (ok) {
        if (!ok) {
          alert('実行できませんでした: ' + path + '\n(ファイル名・リポジトリの公開設定・ページの制限を確認してください)');
        }
      });
  } catch (e) {
    alert('エラー: ' + e);
  }
})();
